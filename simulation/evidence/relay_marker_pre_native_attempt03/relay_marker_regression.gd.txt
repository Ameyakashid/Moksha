func alphas(sonar) -> Array:
	var result: Array=[]
	for record in sonar.geometry: result.append(sonar._segment_alphas(record))
	return result

func mission_test() -> void:
	var host=load("res://demo/mission.tscn").instantiate();root.add_child(host);host.save_enabled=false;host.start_new(true)
	host.set_process(false);host.set_physics_process(false)
	for sector in [0,1,2,4,5]:
		host.load_sector(sector);host.room.process_mode=Node.PROCESS_MODE_DISABLED
		var points:Array=host._relay_points();var first:Vector2=points[0]
		host.player.reset_actor(first,1)
		check(host.relay_states.is_empty() and not host.relay_on,"proximity alone never activates relay")
		for distance in [45.01,45.0]:
			host.player.reset_actor(first+Vector2(-distance,0),1);host.bus.clear_history();host.interact()
			check(host.relay_states.is_empty() and host.bus.event_history.is_empty(),"strict E relay boundary excludes distance at least 45")
		for index in points.size():
			host.player.reset_actor(points[index]+Vector2(-44.99,0),1);host.bus.clear_history();host.interact()
			check(host.relay_states.size()==index+1 and host.relay_states.has(index),"E activates only the next targeted inactive relay")
			var events:Array=host.bus.event_history
			check(events.size()==1 and events[0].origin==points[index]-Vector2(0,20) and events[0].kind==&"prop_impact" and events[0].loudness==1.4 and events[0].source_id==host.room.get_instance_id(),"exact authored-point room-sourced relay noise")
			check(host.relay_on==(index==points.size()-1) and host.room.relay_on==host.relay_on,"global completion only after all relay indices")
			host.room._process(0)
			await process_frame
			for shutter in host.room.shutters:check(shutter.is_open==host.relay_states.has(shutter.relay_index),"only matching actual shutter opens")
			if index<points.size()-1:check(not host.room.gate_open and host._pending_sector==-1,"partial relays cannot advance or open onward")
			host.bus.clear_history();host.interact();check(host.relay_states.size()==index+1 and host.bus.event_history.is_empty(),"repeated E cannot reactivate or emit relay noise")
		check(host.room.gate_open==not host.sectors[sector].get("lock",false),"gate retains enemy-lock policy after relay completion")
		host.load_sector(sector);check(host.relay_states.is_empty() and not host.room.relay_on and host.room.relay_markers.all(func(m):return not m.is_active),"sector retry reconstructs inactive relay state")
	# Explicit synthetic mixed-key fixture: mission prefers plural exclusively.
	var mixed:Dictionary=host.sectors[0].duplicate(true)
	mixed.relays=[Vector2(200,300),Vector2(200,300)];mixed.relay=Vector2(400,300);mixed.enemies=[];mixed.props=[];mixed.erase("cache")
	host.sectors[0]=mixed;host.load_sector(0);host.room.set_process(false)
	check(host._relay_points().size()==2 and host.room.relay_markers.size()==3,"mixed fixture preserves mission plural preference versus painting both keys")
	host.player.reset_actor(Vector2(200,300),1);host.interact();check(host.relay_states==[0] and not host.relay_on,"overlapping relays activate first inactive point only")
	host.interact();check(host.relay_states==[0,1] and host.relay_on and host.room.relay_markers[2].is_active,"next E activates next inactive; global OR paints singular without mission objective")
	mixed.relays=[];host.sectors[0]=mixed;host.load_sector(0)
	check(host._relay_points().is_empty() and host.room.relay_markers.size()==1 and host.room.gate_open,"empty plural key leaves singular paint outside mission objectives")
	paused=false;host.free();await process_frame
