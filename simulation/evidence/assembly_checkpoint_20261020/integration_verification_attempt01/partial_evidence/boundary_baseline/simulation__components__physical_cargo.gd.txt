## Opt-in rigid prop source. Stock visual/launch/noise/physics remain inherited;
## only legacy take_hit admission is replaced. No source-side target teleport.

const ImpactProfile = preload("res://simulation/resources/physical_impact_profile.gd")
@export var impact_profile: ImpactProfile = ImpactProfile.new()

var captured_contacts := 0
var resolved_contacts := 0
var last_impact: Dictionary = {}
var _impact_origin := Vector2.ZERO
var _visited_receivers: Dictionary = {}
var _delivery_scope: Dictionary = {}
var _pending: Dictionary = {}
var _packet_sequence := 0
var _lifecycle := 0
var _reject_through_frame := -1
var _charged_targets: Dictionary = {}
var _pending_charge: Dictionary = {}
var _charge_scope: Dictionary = {}
var charge_transfers := 0
var _break_lifecycle := -1
var _break_target_ref: WeakRef
var _break_receipt: Dictionary = {}
var _receiving_charge := false
var _receiving_strike := false
var _strike_presentation_epoch := -1
var _strike_presentation_token: Dictionary = {}

func _ready() -> void:
	impact_profile = impact_profile.duplicate(true) if impact_profile != null else ImpactProfile.new()
	super._ready()
	max_contacts_reported = maxi(max_contacts_reported, 16)
	_impact_origin = global_position
func _notification(what: int) -> void:
	if what in [NOTIFICATION_PAUSED, NOTIFICATION_UNPAUSED, NOTIFICATION_DISABLED, NOTIFICATION_ENABLED, NOTIFICATION_EXIT_TREE]:
		invalidate_pending_contacts()
func invalidate_pending_contacts() -> void:
	# Call before external reset/reposition or simulation pause. Identity does
	# not rearm; the fence excludes previous-step contacts after a transition.
	_lifecycle += 1
	_delivery_scope.clear()
	_pending.clear()
	_charge_scope.clear()
	_pending_charge.clear()
	_reject_through_frame = Engine.get_physics_frames() + 1
func set_impact_simulation_enabled(value: bool) -> void:
	if simulation_enabled == value: return
	invalidate_pending_contacts()
	simulation_enabled = value
func launch(direction: int, force: float = 1.0) -> void:
	if not _simulating() or not is_finite(force) or force <= 0.0 or direction not in [-1, 1]: return
	invalidate_pending_contacts()
	impact_epoch += 1
	_visited_receivers.clear()
	_charged_targets.clear()
	_delivery_scope.clear()
	_impact_origin = global_position
	super.launch(direction, force)
func _strike_geometry() -> Array:
	var geometry: Array = []
	var active_shapes := 0
	for owner_id in get_shape_owners():
		var shapes: Array = []
		for index in shape_owner_get_shape_count(owner_id):
			var shape := shape_owner_get_shape(owner_id, index)
			if shape == null: return []
			# Server geometry includes in-place edits, not just a reusable RID.
			shapes.append({"id":shape.get_instance_id(),"rid":shape.get_rid(),"data":PhysicsServer2D.shape_get_data(shape.get_rid())})
			if not is_shape_owner_disabled(owner_id): active_shapes += 1
		geometry.append({"owner":owner_id,"transform":shape_owner_get_transform(owner_id),
			"disabled":is_shape_owner_disabled(owner_id),"one_way":is_shape_owner_one_way_collision_enabled(owner_id),"shapes":shapes})
	return geometry if active_shapes > 0 else []
func capture_strike_generation() -> Dictionary:
	if not _simulating() or collision_layer == 0: return {}
	var geometry := _strike_geometry()
	if geometry.is_empty(): return {}
	var token := {"host_id":get_instance_id(),"rid":get_rid(),"world":get_world_2d().space,
		"lifecycle":_lifecycle,"epoch":impact_epoch,"layer":collision_layer,"mask":collision_mask,
		"node_transform":global_transform,"physics_transform":PhysicsServer2D.body_get_state(get_rid(),PhysicsServer2D.BODY_STATE_TRANSFORM),"geometry":geometry}
	token.make_read_only()
	return token
func strike_generation_is_current(token: Dictionary) -> bool:
	return not token.is_empty() and token == capture_strike_generation()
func _capture_launch_presentation() -> Variant:
	var token := capture_strike_generation()
	if _receiving_strike and impact_epoch == _strike_presentation_epoch:
		_strike_presentation_token = token
	return token
func _launch_presentation_is_current(token: Variant) -> bool:
	return token is Dictionary and strike_generation_is_current(token)
func receive_strike(packet: Dictionary) -> Dictionary:
	if _receiving_strike or not _simulating(): return {"accepted":false,"reason":&"unavailable"}
	var token: Variant = packet.get("receiver_generation")
	var source: Variant = packet.get("source")
	var adapter: Variant = packet.get("contact_adapter")
	var direction: Variant = packet.get("facing")
	var scale: Variant = packet.get("prop_launch_scale")
	if not token is Dictionary or not strike_generation_is_current(token): return {"accepted":false,"reason":&"stale_geometry"}
	if not direction is int or direction not in [-1,1] or not (scale is float or scale is int) or not is_finite(float(scale)) or float(scale) <= 0: return {"accepted":false,"reason":&"invalid_launch"}
	if not is_instance_valid(source) or not source is Node2D or not source.is_inside_tree() or source.is_queued_for_deletion() or source.get_world_2d() != get_world_2d(): return {"accepted":false,"reason":&"invalid_source"}
	if not is_instance_valid(adapter) or not adapter is MokshaContactAdapter or not source.is_ancestor_of(adapter) or adapter.actor != source: return {"accepted":false,"reason":&"invalid_authority"}
	_receiving_strike = true
	var permit: bool = adapter.contact_is_current(packet,self)
	if not permit or not strike_generation_is_current(token):
		_receiving_strike = false
		return {"accepted":false,"reason":&"stale_permit"}
	var before := hit_count
	var launched_epoch := impact_epoch + 1
	var source_id: int = source.get_instance_id()
	_strike_presentation_epoch = launched_epoch
	_strike_presentation_token = {}
	launch(direction,float(scale))
	var committed_token := _strike_presentation_token
	_strike_presentation_token = {}
	_strike_presentation_epoch = -1
	_receiving_strike = false
	# Launch is one completed physical operation; it retires the adapter's old
	# geometry token. Do not issue a replacement permit to continue that sweep.
	var receipt := {"accepted":hit_count > before,"retired":true,"source_id":source_id,
		"host_id":get_instance_id(),"serial":packet.get("serial",-1),"epoch":launched_epoch,
		"direction":direction,"scale":float(scale),"physics_frame":Engine.get_physics_frames()}
	receipt.make_read_only()
	if strike_generation_is_current(committed_token) and impact_epoch == launched_epoch and _lifecycle == int(token.lifecycle)+1:
		strike_launch_resolved.emit(receipt)
	return receipt

func _physics_process(dt: float) -> void:
	var charged := charge > 0.0
	super._physics_process(dt)
	if charged and charge <= 0.0: invalidate_pending_contacts()
func _simulating() -> bool:
	return alive and simulation_enabled and is_inside_tree() and not is_queued_for_deletion() and can_process() \
		and is_physics_processing() and not get_tree().paused and not freeze
func _available() -> bool:
	return _simulating() and impact_profile != null and impact_profile.enabled and impact_profile.valid() \
		and (not impact_profile.requires_charge or (is_finite(charge) and charge > 0.0))
func _can_transfer_charge() -> bool:
	return _simulating() and impact_profile != null and impact_profile.transfers_charge \
		and is_finite(charge) and charge > 0.0 and is_finite(tuning.chain_speed) and tuning.chain_speed > 0.0 \
		and is_finite(tuning.chain_charge_seconds) and tuning.chain_charge_seconds > 0.0
func _integrate_forces(state: PhysicsDirectBodyState2D) -> void:
	if _can_transfer_charge() and Engine.get_physics_frames() > _reject_through_frame:
		_capture_charge_contacts(state)
	if _available() and Engine.get_physics_frames() > _reject_through_frame:
		for index in state.get_contact_count():
			var target: Variant = state.get_contact_collider_object(index)
			if not is_instance_valid(target) or not target is PhysicsBody2D or target.is_queued_for_deletion(): continue
			var receiver := Receiver.registered(target)
			if receiver == null: continue
			var id: int = target.get_instance_id()
			if _visited_receivers.has(id): continue
			var normal := state.get_contact_local_normal(index).normalized()
			var relative := state.get_contact_local_velocity_at_position(index) - state.get_contact_collider_velocity_at_position(index)
			var speed := maxf(0.0, -relative.dot(normal))
			if not is_finite(speed) or speed < impact_profile.minimum_normal_speed: continue
			var generation: int = receiver.capture_generation()
			if generation < 0: continue
			_visited_receivers[id] = true
			_packet_sequence += 1
			captured_contacts += 1
			var packet := {"source": self, "epoch": impact_epoch, "packet_id": _packet_sequence,
				"target_generation": generation, "receiver_binding": receiver._binding, "lifecycle": _lifecycle, "origin": _impact_origin,
				"contact_position": state.get_contact_collider_position(index), "normal": normal,
				"incoming_velocity": relative, "source_contact_velocity": state.get_contact_local_velocity_at_position(index),
				"target_contact_velocity": state.get_contact_collider_velocity_at_position(index), "normal_speed": speed,
				"impulse": -normal * impact_profile.response_speed + Vector2(0, impact_profile.vertical_bias),
				"amount": impact_profile.damage, "stun_seconds": impact_profile.stun_seconds, "hitpause": impact_profile.hitpause_seconds,
				"blockable": impact_profile.blockable, "parryable": impact_profile.parryable, "rollable": impact_profile.rollable,
				"physics_frame": Engine.get_physics_frames()}
			packet.make_read_only()
			_pending[_packet_sequence] = {"target": weakref(target), "receiver": weakref(receiver), "packet": packet}
			_deliver.call_deferred(_pending[_packet_sequence].target, _pending[_packet_sequence].receiver, packet)
	_impact_origin = state.transform.origin
	super._integrate_forces(state)
func physical_impact_is_current(packet: Dictionary, target: Node) -> bool:
	return _available() and not _delivery_scope.is_empty() and packet == _delivery_scope.packet \
		and packet.epoch == impact_epoch and packet.lifecycle == _lifecycle and _delivery_scope.target.get_ref() == target \
		and is_instance_valid(target) and target.is_inside_tree() and not target.is_queued_for_deletion() \
		and target.get_world_2d() == get_world_2d()
func _deliver(target_ref: WeakRef, receiver_ref: WeakRef, packet: Dictionary) -> void:
	var id: Variant = packet.get("packet_id")
	if not id is int or not _pending.has(id): return
	var captured: Dictionary = _pending[id]
	if packet != captured.packet or target_ref.get_ref() != captured.target.get_ref() or receiver_ref.get_ref() != captured.receiver.get_ref(): return
	_pending.erase(id)
	if not _available() or packet.epoch != impact_epoch or packet.lifecycle != _lifecycle: return
	var target: Variant = target_ref.get_ref()
	var receiver: Variant = receiver_ref.get_ref()
	if not is_instance_valid(target) or not is_instance_valid(receiver) or Receiver.registered(target) != receiver: return
	_delivery_scope = {"target": target_ref, "packet": packet}
	var result: Dictionary = receiver.receive_physical_impact(packet)
	_delivery_scope.clear()
	if not _available() or packet.epoch != impact_epoch or packet.lifecycle != _lifecycle or bool(result.get("retired", false)): return
	# Receiver receipt callbacks may retire the completed target life/binding.
	# Its value-only receipt is already committed; no source feedback may follow.
	if bool(result.get("accepted", false)) and not _feedback_current(target_ref, packet, result): return
	if bool(result.get("accepted", false)): resolved_contacts += 1
	last_impact = result.duplicate(true)
	var receipt := result.duplicate(true)
	receipt.make_read_only()
	physical_contact_resolved.emit(receipt)
	if _available() and bool(result.get("accepted", false)) and impact_profile.break_on_accepted_impact:
		_accepted_impact_feedback(target_ref, packet, result)

func _feedback_current(target_ref: WeakRef, packet: Dictionary, result: Dictionary) -> bool:
	var target: Variant = target_ref.get_ref()
	if not (_available() and packet.epoch == impact_epoch and packet.lifecycle == _lifecycle \
		and is_instance_valid(target) and target.is_inside_tree() and not target.is_queued_for_deletion() \
		and target.get_world_2d() == get_world_2d()): return false
	var receiver := Receiver.registered(target)
	return receiver != null and receiver._binding == result.get("receiver_binding") \
		and is_instance_valid(receiver._provider()) and receiver._provider().damage_generation() == result.get("target_generation")
func _accepted_impact_feedback(target_ref: WeakRef, packet: Dictionary, result: Dictionary) -> void:
	if not _feedback_current(target_ref, packet, result): return
	var direction := signf(Vector2(packet.impulse).x)
	if direction == 0: direction = 1.0
	# Legacy presentation signal carries the authored integer amount. Exact
	# Health-clamped/fractional damage is in physical_contact_resolved instead.
	impacted.emit(self, target_ref.get_ref(), int(packet.amount), direction)
	if not _feedback_current(target_ref, packet, result): return
	shake_requested.emit(6.0)
	if not _feedback_current(target_ref, packet, result): return
	burst_requested.emit(global_position, 20, Color("eab779"), 130.0)
	if not _feedback_current(target_ref, packet, result): return
	_break_target_ref = target_ref
	_break_receipt = result
	_break_apart()
func _break_apart() -> void:
	if not alive: return
	invalidate_pending_contacts()
	_break_lifecycle = _lifecycle
	super._break_apart()
	_break_target_ref = null
	_break_receipt = {}
func _break_presentation_is_current() -> bool:
	if _break_lifecycle != _lifecycle or not is_inside_tree() or is_queued_for_deletion() or not simulation_enabled or get_tree().paused: return false
	if _break_target_ref == null: return true
	var target: Variant = _break_target_ref.get_ref()
	if not is_instance_valid(target) or not target.is_inside_tree() or target.is_queued_for_deletion() or target.get_world_2d() != get_world_2d(): return false
	var receiver := Receiver.registered(target)
	return receiver != null and receiver._binding == _break_receipt.get("receiver_binding") \
		and is_instance_valid(receiver._provider()) and receiver._provider().damage_generation() == _break_receipt.get("target_generation")

func _capture_charge_contacts(state: PhysicsDirectBodyState2D) -> void:
	for index in state.get_contact_count():
		var target: Variant = state.get_contact_collider_object(index)
		if not is_instance_valid(target) or not target is RigidBody2D or not target.has_method("receive_physical_charge"): continue
		if not target._simulating() or target.get_world_2d() != get_world_2d(): continue
		var id: int = target.get_instance_id()
		if _charged_targets.has(id): continue
		var normal := state.get_contact_local_normal(index).normalized()
		var source_velocity := state.get_contact_local_velocity_at_position(index)
		var target_velocity := state.get_contact_collider_velocity_at_position(index)
		var speed := maxf(0.0, -(source_velocity - target_velocity).dot(normal))
		if not is_finite(speed) or speed <= tuning.chain_speed: continue
		_charged_targets[id] = weakref(target)
		_packet_sequence += 1
		var packet := {"source": self, "epoch": impact_epoch, "packet_id": _packet_sequence, "lifecycle": _lifecycle,
			"target_lifecycle": target._lifecycle, "target_epoch": target.impact_epoch, "seconds": tuning.chain_charge_seconds,
			"normal_speed": speed, "normal": normal, "contact_position": state.get_contact_collider_position(index),
			"source_contact_velocity": source_velocity, "target_contact_velocity": target_velocity,
			"physics_frame": Engine.get_physics_frames()}
		packet.make_read_only()
		_pending_charge[_packet_sequence] = {"target": weakref(target), "packet": packet}
		_deliver_charge.call_deferred(_pending_charge[_packet_sequence].target, packet)
func physical_charge_is_current(packet: Dictionary, target: Node) -> bool:
	return _can_transfer_charge() and not _charge_scope.is_empty() and packet == _charge_scope.packet \
		and packet.epoch == impact_epoch and packet.lifecycle == _lifecycle and _charge_scope.target.get_ref() == target \
		and is_instance_valid(target) and target.is_inside_tree() and not target.is_queued_for_deletion() \
		and target.get_world_2d() == get_world_2d()
func _deliver_charge(target_ref: WeakRef, packet: Dictionary) -> void:
	var id: Variant = packet.get("packet_id")
	if not id is int or not _pending_charge.has(id): return
	var captured: Dictionary = _pending_charge[id]
	if packet != captured.packet or target_ref.get_ref() != captured.target.get_ref(): return
	_pending_charge.erase(id)
	if not _can_transfer_charge() or packet.epoch != impact_epoch or packet.lifecycle != _lifecycle: return
	var target: Variant = target_ref.get_ref()
	if not is_instance_valid(target) or not target.has_method("receive_physical_charge"): return
	_charge_scope = {"target": target_ref, "packet": packet}
	var accepted: bool = target.receive_physical_charge(packet)
	_charge_scope.clear()
	if not accepted or not _can_transfer_charge() or packet.epoch != impact_epoch or packet.lifecycle != _lifecycle: return
	charge_transfers += 1
	var receipt := {"source_id": get_instance_id(), "target_id": target.get_instance_id(), "epoch": packet.epoch,
		"target_epoch": packet.target_epoch, "seconds": packet.seconds, "normal_speed": packet.normal_speed,
		"contact_position": packet.contact_position, "physics_frame": packet.physics_frame}
	receipt.make_read_only()
	physical_charge_resolved.emit(receipt)
func receive_physical_charge(packet: Dictionary) -> bool:
	if _receiving_charge: return false
	_receiving_charge = true
	var accepted := _consume_physical_charge(packet)
	_receiving_charge = false
	return accepted
func _consume_physical_charge(packet: Dictionary) -> bool:
	if not _simulating(): return false
	var source: Variant = packet.get("source")
	if not is_instance_valid(source) or source == self or not source.has_method("physical_charge_is_current"): return false
	if packet.get("target_lifecycle") != _lifecycle or packet.get("target_epoch") != impact_epoch: return false
	var seconds: Variant = packet.get("seconds")
	if not (seconds is float or seconds is int) or not is_finite(float(seconds)) or float(seconds) <= 0.0: return false
	var lifecycle := _lifecycle
	var epoch := impact_epoch
	if not source.physical_charge_is_current(packet, self): return false
	if not _simulating() or lifecycle != _lifecycle or epoch != impact_epoch: return false
	# Transfer does not launch, reset visited receivers, advance an epoch, or
	# synthesize motion. The solver alone exchanges the bodies' momentum.
	charge = maxf(charge, float(seconds))
	return true
