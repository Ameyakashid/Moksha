extends "res://simulation/components/incoming_strike_receiver.gd"
var after_capture: Callable
var before_receive: Callable
var captured_packet: Dictionary = {}
func capture_strike_generation() -> Dictionary:
	var token := super.capture_strike_generation()
	if not token.is_empty() and after_capture.is_valid(): after_capture.call()
	return token
func receive_strike(packet: Dictionary) -> Dictionary:
	captured_packet = packet.duplicate(true)
	if before_receive.is_valid(): before_receive.call(packet)
	return super.receive_strike(packet)
