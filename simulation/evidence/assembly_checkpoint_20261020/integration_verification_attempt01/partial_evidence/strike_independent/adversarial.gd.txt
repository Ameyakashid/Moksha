extends "res://simulation/tests/incoming_strike_regression.gd"
const Probe = preload("/tmp/moksha-strike-independent/receiver_probe.gd")
func _probe(actor: Node) -> Node:
	var old = _receiver(actor)
	old.unbind()
	old.get_parent().remove_child(old)
	old.queue_free()
	var probe = ReceiverScene.instantiate()
	probe.set_script(Probe)
	actor.get_node("BodyShape").add_child(probe)
	probe.strike_resolved.connect(func(receipt): receipts.append(receipt.duplicate(true)))
	return probe
func _run() -> void:
	for stage in ["capture", "confirm"]:
		for operation in ["shape_missing", "shape_queued", "receiver_removed", "source_removed"]:
			await _begin(stage + "/" + operation, "raw", "raw")
			var probe = _probe(right)
			var called: Array[bool] = []
			var act := func():
				if not called.is_empty(): return
				called.append(true)
				match operation:
					"shape_missing":
						var shape = probe.get_node("Shape")
						probe.remove_child(shape)
						shape.queue_free()
					"shape_queued": probe.get_node("Shape").queue_free()
					"receiver_removed":
						probe.get_parent().remove_child(probe)
					"source_removed": world.remove_child(left)
			if stage == "capture": probe.after_capture = act
			else: left.movement_event.connect(func(kind,_payload):
				if kind == &"strike_confirmed": act.call())
			await _frames(3)
			await _attack(left)
			await _frames(ceili(hz*.3))
			_check(not called.is_empty(), "hook ran")
			_check(_health(right) == 30 and receipts.is_empty(), "retired capture has no target response")
			if operation == "source_removed": left.queue_free()
			if operation == "receiver_removed": probe.queue_free()
			await _end()
	for mode in ["paused_only", "pause_cycle", "receiver_disable", "source_disable"]:
		await _begin("ALWAYS/" + mode, "raw", "raw")
		left.process_mode = Node.PROCESS_MODE_ALWAYS
		right.process_mode = Node.PROCESS_MODE_ALWAYS
		var called: Array[bool] = []
		var observed: Array[bool] = []
		left.movement_event.connect(func(kind,_payload):
			if kind != &"strike_confirmed" or not called.is_empty(): return
			called.append(true)
			match mode:
				"paused_only": paused = true
				"pause_cycle":
					paused = true
					paused = false
				"receiver_disable": right.process_mode = Node.PROCESS_MODE_DISABLED
				"source_disable": left.process_mode = Node.PROCESS_MODE_DISABLED
		)
		left.strike_sample.connect(func(_source,_packet):
			if not called.is_empty() and observed.is_empty(): observed.append(_health(right) == 30))
		await _attack(left)
		await _frames(ceili(hz*.25))
		_check(not called.is_empty(), "hook ran")
		print("ALWAYS detail ",mode," observed=",observed," health=",_health(right))
		_check(observed == [true] and _health(right) == 30, "global pause or disabled owner rejects target response")
		paused = false
		await _end()
	await _begin("malformed and reentrant permit", "raw", "raw")
	var probe = _probe(right)
	await _frames(3)
	var base := {"serial":1,"source_epoch":0,"contact_ticket":1,"origin":Vector2.ZERO,"source_position":Vector2.ZERO,"contact_position":Vector2.ZERO,"impulse":Vector2.ZERO,"damage":1.0}
	for field in ["serial", "source_epoch", "contact_ticket", "origin", "source_position", "contact_position", "impulse", "damage"]:
		var bad = base.duplicate(true)
		bad[field] = "wrong type"
		_check(not probe.receive_strike(bad).accepted and _health(right) == 30, "reject malformed " + field)
	_check(not probe.receive_strike(base).accepted, "valid fields without scope reject")
	var nested: Array[StringName] = []
	right.movement_event.connect(func(kind,_payload):
		if kind == &"hurt_started": nested.append(probe.receive_strike(probe.captured_packet).reason))
	await _attack(left)
	_check(await _until(func(): return _health(right) == 29), "normal contact still accepted")
	_check(nested == [&"reentrant"] and receipts.size() == 1, "nested permitted packet cannot duplicate committed operation")
	var copied = probe.captured_packet.duplicate(true)
	_check(not probe.receive_strike(copied).accepted and _health(right) == 29, "copied packet cannot outlive dispatch")
	await _end()
	for field in ["size", "previous_center", "origin", "damage"]:
		await _begin("canonical forge " + field, "raw", "raw")
		var adapter = left.get_node("ContactAdapter")
		adapter.bind_actor(null)
		adapter.actor = left
		var called: Array[bool] = []
		left.strike_sample.connect(func(source,packet):
			if not packet.eligible: return
			called.append(true)
			var bad = packet.duplicate(true)
			bad[field] = 1e30 if field == "damage" else Vector2(1e30,1e30)
			adapter.sample_strike(source,bad))
		await _attack(left)
		await _frames(ceili(hz*.3))
		_check(not called.is_empty() and _health(right) == 30 and adapter.confirmed_contacts == 0, "large finite forgery rejects before sweep")
		await _end()
	print("INDEPENDENT STRIKE ADVERSARIAL %d Hz: %d scenarios, %d checks, %d failures" % [hz,scenarios.size(),checks,failures.size()])
	for failure in failures: printerr(failure)
	var report := {"hz":hz,"scenarios":scenarios,"checks":checks,"failures":failures}
	FileAccess.open("/tmp/moksha-strike-independent/adversarial_%d.json" % hz,FileAccess.WRITE).store_string(JSON.stringify(report,"  "))
	quit(0 if failures.is_empty() else 1)
