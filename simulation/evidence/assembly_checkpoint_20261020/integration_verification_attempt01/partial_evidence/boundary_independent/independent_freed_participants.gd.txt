extends "res://simulation/tests/boundary_regression.gd"
func _run() -> void:
	await begin("independent freed participant public APIs");hazard.enabled=false;spawn_actor()
	var receiver=actor.get_node("BoundaryReceiver")
	var expired=Node2D.new();expired.free()
	var store=MokshaHealthComponent.new();store.free()
	check(MokshaBoundaryReceiver.registered(expired)==null,"freed registered host rejects")
	check(not receiver.bind(expired,health),"freed bind host rejects")
	check(not receiver.bind(actor,store) and receiver.get_host()==actor and receiver.get_health()==health,"freed bind Health rejects without losing installed binding")
	check(not receiver.boundary_attack_is_current({},expired),"freed attack host rejects")
	check(not receiver.claim_cargo_retirement({},expired),"freed cargo claim host rejects")
	check(not hazard.boundary_packet_is_current({},expired,actor),"freed packet receiver rejects")
	check(not hazard.boundary_packet_is_current({},receiver,expired),"freed packet host rejects")
	check(not hazard.hold_guard({},receiver,actor,store,{}),"freed hold Health rejects")
	for who in ["owner","source","scope"]:
		var token=health.reserve_guarded_debit(expired if who=="owner" else receiver,expired if who=="source" else hazard,expired if who=="scope" else world,actor.damage_generation(),actor.damage_admission_sequence()+1)
		check(token.is_empty(),"freed reservation "+who+" rejects")
	check(not receiver.receive_boundary({"source":expired}).resolved,"freed packet source rejects")
	var box=spawn_cargo()
	for invalid in [null,42,"wrong",expired]:check(not box.retire_from_boundary({},invalid),"invalid cargo authority rejects "+str(typeof(invalid)))
	world.queue_free();await frames(3)
	FileAccess.open("/tmp/moksha_boundary_independent/freed_participants.json",FileAccess.WRITE).store_string(JSON.stringify({"checks":checks,"failures":failures},"  ")+"\n")
	print("FREED PARTICIPANTS: %d checks, %d failures"%[checks.size(),failures.size()])
	quit(0 if failures.is_empty() else 1)
